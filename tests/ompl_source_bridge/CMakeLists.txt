cmake_minimum_required(VERSION 3.20)
project(hybrid_astar_ompl_source_bridge LANGUAGES CXX)
enable_testing()

# Standalone, manual-only integration verification against OMPL source code.
# NOT a replacement for a complete OMPL-ON package build or installation.
# Usage: cmake -S tests/ompl_source_bridge -B build-source-ompl \
#    -DOMPL_SOURCE_ROOT=/path/to/ompl-2.0.2 -DEIGEN3_INCLUDE_DIR=/path/to/eigen3
set(OMPL_SOURCE_ROOT "" CACHE PATH "Unpacked OMPL 2.0.2 source tree")
set(EIGEN3_INCLUDE_DIR "" CACHE PATH "Directory containing Eigen/Core")
if(NOT EXISTS "${OMPL_SOURCE_ROOT}/src/ompl/base/spaces/src/ReedsSheppStateSpace.cpp")
  message(FATAL_ERROR "Set OMPL_SOURCE_ROOT to unpacked OMPL sources")
endif()
if(NOT EXISTS "${EIGEN3_INCLUDE_DIR}/Eigen/Core")
  message(FATAL_ERROR "Set EIGEN3_INCLUDE_DIR to path containing Eigen/Core")
endif()

set(HYBRID_ASTAR_ENABLE_OMPL OFF CACHE BOOL "OFF: source subset used here" FORCE)
set(HYBRID_ASTAR_BUILD_BENCHMARKS OFF CACHE BOOL "No benchmarks in focused smoke" FORCE)
set(HYBRID_ASTAR_BUILD_TESTS ON CACHE BOOL "Build all core tests" FORCE)
set(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
add_subdirectory("${REPO_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/project")
set(OMPL_SRC "${OMPL_SOURCE_ROOT}/src")

add_library(ompl_rs_source_subset STATIC
  ${OMPL_SRC}/ompl/base/src/StateSpace.cpp
  ${OMPL_SRC}/ompl/base/src/ProjectionEvaluator.cpp
  ${OMPL_SRC}/ompl/base/src/StateSampler.cpp
  ${OMPL_SRC}/ompl/base/src/GenericParam.cpp
  ${OMPL_SRC}/ompl/util/src/Console.cpp
  ${OMPL_SRC}/ompl/util/src/String.cpp
  ${OMPL_SRC}/ompl/util/src/RandomNumbers.cpp
  ${OMPL_SRC}/ompl/base/spaces/src/RealVectorBounds.cpp
  ${OMPL_SRC}/ompl/base/spaces/src/RealVectorStateProjections.cpp
  ${OMPL_SRC}/ompl/base/spaces/src/RealVectorStateSpace.cpp
  ${OMPL_SRC}/ompl/base/spaces/src/SO2StateSpace.cpp
  ${OMPL_SRC}/ompl/base/spaces/src/SE2StateSpace.cpp
  ${OMPL_SRC}/ompl/base/spaces/src/ReedsSheppStateSpace.cpp)
target_compile_features(ompl_rs_source_subset PUBLIC cxx_std_20)
target_include_directories(ompl_rs_source_subset PUBLIC
  "${OMPL_SRC}" "${EIGEN3_INCLUDE_DIR}")

# This header is normally generated by the full OMPL CMake package build.
# The selected state-space sources use only these feature/version constants.
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/ompl_generated/ompl")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ompl_generated/ompl/config.h"
  "#pragma once\n#define OMPL_VERSION \"2.0.2\"\n#define OMPL_MAJOR_VERSION 2\n#define OMPL_MINOR_VERSION 0\n#define OMPL_PATCH_VERSION 2\n#define OMPL_HAVE_FLANN 0\n#define OMPL_HAVE_SPOT 0\n#define OMPL_HAVE_NUMPY 0\n")
target_include_directories(ompl_rs_source_subset PUBLIC
  "${CMAKE_CURRENT_BINARY_DIR}/ompl_generated")

if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND NOT APPLE)
  target_compile_options(ompl_rs_source_subset PRIVATE
    -ffunction-sections -fdata-sections)
endif()

add_library(hybrid_astar_source_ompl STATIC
  "${REPO_ROOT}/src/steering/ompl_reeds_shepp.cpp")
target_compile_features(hybrid_astar_source_ompl PUBLIC cxx_std_20)
target_link_libraries(hybrid_astar_source_ompl PUBLIC
  hybrid_astar_core ompl_rs_source_subset)

foreach(test_name ompl_smoke ompl_integration_test)
  add_executable(ompl_source_${test_name}
    "${REPO_ROOT}/tests/${test_name}.cpp")
  target_link_libraries(ompl_source_${test_name} PRIVATE
    hybrid_astar_source_ompl)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang" AND NOT APPLE)
    target_link_options(ompl_source_${test_name} PRIVATE -Wl,--gc-sections)
  endif()
  add_test(NAME source_${test_name} COMMAND ompl_source_${test_name})
endforeach()
